  Collapse only when the two land on the same Secret in the same namespace;

  subtree being absent.

{{- $certManager := (index .Values "nico-rest-cert-manager") | default dict }}

{{- $trustedSecret := ($siteManager.credsManagerCa | default dict).secretName | default "" }}
{{- $signingNamespace := $certManager.namespaceOverride | default .Release.Namespace }}
{{- $trustedNamespace := $siteManager.namespaceOverride | default .Release.Namespace }}

{{- if $certManager.enabled }}

{{- if $siteManager.enabled }}
